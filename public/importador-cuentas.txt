// 1) Ejecutá esta función una sola vez: trae los últimos 60 días y deja programado el resto.
function activar() {
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('importarCuentas').timeBased().everyDays(5).atHour(8).create();
  importarCuentas(60);
}

// 2) Se ejecuta sola cada 5 días.
function importarCuentas(dias) {
  dias = typeof dias === 'number' ? dias : 12;
  const movimientos = [];
  const resumenes = [];
  const lee = function (id) { return LEER.indexOf(id) >= 0; };

  if (lee('uala')) {
    mails('from:uala.com.ar subject:"transferencia fue exitosa"', dias, function (m, texto) {
      const monto = texto.match(/Monto\s*\$\s*([\d.,]+)/i);
      const destino = texto.match(/Destinatario\s+(.+?)\s+(CBU|CVU|Alias|Id\.)/i);
      const operacion = texto.match(/Id\.?\s*Op\.?\s*([\w-]+)/i);
      const nombre = destino ? destino[1].trim() : '';
      if (!monto || !operacion || esPropio(nombre)) return;
      movimientos.push(gasto('uala:' + operacion[1], m, numero(monto[1]), nombre ? 'Transferencia a ' + titulo(nombre) : 'Transferencia', 'transferencias', 'uala'));
    });
    mails('from:uala.com.ar subject:"Resumen tarjeta"', dias, function (m, texto) {
      const vence = texto.match(/vencimiento\s+(\d{1,2})\s+([A-Za-zÁÉÍÓÚáéíóú]{3})[A-Za-zÁÉÍÓÚáéíóú]*\.?\s+(\d{4})/i);
      const pesos = texto.match(/Total en pesos\s*\$\s*([\d.,]+)/i);
      const dolares = texto.match(/Total en d[oó]lares\s*(?:USD|US\$)\s*([\d.,]+)/i);
      const mes = vence ? MESES[vence[2].toLowerCase()] : null;
      if (!vence || !pesos || !mes) return;
      resumenes.push({ provider: 'uala', card: 'Ualá', color: '#2F80ED', month: vence[3] + '-' + mes, due_date: vence[3] + '-' + mes + '-' + ('0' + vence[1]).slice(-2), total: numero(pesos[1]), total_usd: dolares ? numero(dolares[1]) : 0 });
    });
  }

  if (lee('mercadopago')) {
    mails('from:mercadopago subject:"Pago aprobado"', dias, function (m, texto) {
      const monto = texto.match(/Pagaste\s*\$\s*([\d.,]+)/i);
      if (!monto) return;
      const comercio = m.getSubject().replace(/^.*Pago aprobado (en|a)\s*/i, '').trim();
      movimientos.push(gasto('mail:' + m.getId(), m, numero(monto[1]), (comercio || 'Pago') + ' (Mercado Pago)', 'otro', 'mercadopago'));
    });
  }

  if (lee('nacion')) {
    mails('from:bna subject:"Aviso de transferencia"', dias, function (m, texto) {
      const datos = texto.match(/transferencia a (.+?) por \$\s*([\d.,]+)/i);
      if (!datos || esPropio(datos[1])) return;
      movimientos.push(gasto('mail:' + m.getId(), m, numero(datos[2]), 'Transferencia a ' + titulo(datos[1].trim()) + ' (Nación)', 'transferencias', 'nacion'));
    });
  }

  if (lee('macro')) {
    mails('from:macro subject:"Aviso de Transferencia"', dias, function (m, texto) {
      const datos = texto.match(/transferencia al CBU\/CVU\/alias\s+(\S+)\s+por\s+\$\s*([\d.,]+)/i);
      if (!datos || esPropio(datos[1])) return;
      movimientos.push(gasto('mail:' + m.getId(), m, numero(datos[2]), 'Transferencia a ' + datos[1] + ' (Macro)', 'transferencias', 'macro'));
    });
  }

  if (lee('wallbit')) {
    const cotizacion = dolar();
    if (cotizacion) {
      mails('from:wallbit subject:Recibiste', dias, function (m, texto) {
        const monto = texto.match(/Monto:\s*USD\s*\$?\s*([\d.,]+)/i);
        const origen = texto.match(/Origen:\s*(.+?)\s*(ID de operaci|$)/i);
        const operacion = texto.match(/ID de operaci[oó]n:\s*([\w-]+)/i);
        if (!monto) return;
        const dolares = Number(monto[1].replace(/,/g, ''));
        const fecha = m.getDate();
        movimientos.push({
          id: operacion ? 'wallbit:' + operacion[1] : 'mail:' + m.getId(),
          type: 'ingreso', category: 'otro', source: 'wallbit', currency: 'USD',
          original: dolares, rate: cotizacion, amount: Math.round(dolares * cotizacion * 100) / 100,
          description: origen ? 'Wallbit · ' + titulo(origen[1]) : 'Wallbit',
          date: Utilities.formatDate(fecha, 'America/Argentina/Buenos_Aires', 'yyyy-MM-dd'), datetime: fecha.toISOString()
        });
      });
    }
  }

  const respuesta = UrlFetchApp.fetch(APP_URL + '/rest/v1/rpc/import_cuentas', {
    method: 'post',
    contentType: 'application/json',
    headers: { apikey: APP_KEY },
    payload: JSON.stringify({ p_token: MI_CODIGO, p_movements: movimientos, p_statements: resumenes }),
    muteHttpExceptions: true
  });
  Logger.log('Encontré ' + movimientos.length + ' movimientos y ' + resumenes.length + ' resúmenes. Respuesta: ' + respuesta.getResponseCode() + ' ' + respuesta.getContentText());
}

function mails(busqueda, dias, alLeer) {
  GmailApp.search(busqueda + ' newer_than:' + dias + 'd', 0, 300).forEach(function (hilo) {
    hilo.getMessages().forEach(function (m) { alLeer(m, m.getPlainBody().replace(/\s+/g, ' ')); });
  });
}

function gasto(id, m, monto, descripcion, categoria, origen) {
  const fecha = m.getDate();
  return { id: id, type: 'gasto', amount: monto, description: descripcion, category: categoria, source: origen, currency: 'ARS', date: Utilities.formatDate(fecha, 'America/Argentina/Buenos_Aires', 'yyyy-MM-dd'), datetime: fecha.toISOString() };
}

// Transferencias a vos mismo (por nombre o alias) no cuentan como gasto.
function esPropio(texto) {
  const palabras = MI_NOMBRE.toLowerCase().split(/\s+/).filter(function (p) { return p.length > 2; });
  if (palabras.length < 2) return false;
  const t = texto.toLowerCase().replace(/[^a-záéíóúñ]/g, '');
  return t.indexOf(palabras[0]) >= 0 && t.indexOf(palabras[palabras.length - 1]) >= 0;
}

// Dólar cripto (compra), el más parecido a lo que te dan en Wallbit. Si no responde, el MEP.
function dolar() {
  const tipos = ['cripto', 'bolsa'];
  for (var i = 0; i < tipos.length; i++) {
    try {
      const datos = JSON.parse(UrlFetchApp.fetch('https://dolarapi.com/v1/dolares/' + tipos[i], { muteHttpExceptions: true }).getContentText());
      if (datos && datos.compra) return Number(datos.compra);
    } catch (e) { }
  }
  return null;
}

const MESES = { ene: '01', jan: '01', feb: '02', mar: '03', abr: '04', apr: '04', may: '05', jun: '06', jul: '07', ago: '08', aug: '08', sep: '09', set: '09', oct: '10', nov: '11', dic: '12', dec: '12' };

// "$50.000,00" o "$3500" → 50000 / 3500
function numero(texto) {
  const limpio = texto.indexOf(',') >= 0 ? texto.replace(/\./g, '').replace(',', '.') : texto.replace(/\.(?=\d{3}(\D|$))/g, '');
  return Number(limpio);
}

function titulo(texto) {
  return texto.toLowerCase().replace(/(^|\s)\S/g, function (letra) { return letra.toUpperCase(); });
}
