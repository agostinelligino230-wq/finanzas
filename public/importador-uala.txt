// 1) Ejecutá esta función una sola vez: trae los últimos 60 días y deja programado el resto.
function activar() {
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('importarUala').timeBased().everyDays(5).atHour(8).create();
  importarUala(60);
}

// 2) Se ejecuta sola cada 5 días.
function importarUala(dias) {
  dias = typeof dias === 'number' ? dias : 12;
  const busqueda = 'from:uala.com.ar newer_than:' + dias + 'd (subject:"transferencia fue exitosa" OR subject:"Resumen tarjeta")';
  const transferencias = [];
  const resumenes = [];
  GmailApp.search(busqueda, 0, 300).forEach(function (hilo) {
    hilo.getMessages().forEach(function (mensaje) {
      const asunto = mensaje.getSubject();
      const texto = mensaje.getPlainBody().replace(/\s+/g, ' ');
      if (/transferencia fue exitosa/i.test(asunto)) {
        const monto = texto.match(/Monto\s*\$\s*([\d.,]+)/i);
        const destino = texto.match(/Destinatario\s+(.+?)\s+(CBU|CVU|Alias|Id\.)/i);
        const operacion = texto.match(/Id\.?\s*Op\.?\s*([\w-]+)/i);
        if (!monto || !operacion) return;
        const nombre = destino ? destino[1].trim() : '';
        if (NO_CONTAR.some(function (n) { return nombre.toLowerCase().indexOf(n) >= 0; })) return;
        const fecha = mensaje.getDate();
        transferencias.push({
          id: 'uala:' + operacion[1],
          amount: numero(monto[1]),
          description: nombre ? 'Transferencia a ' + titulo(nombre) : 'Transferencia',
          date: Utilities.formatDate(fecha, 'America/Argentina/Buenos_Aires', 'yyyy-MM-dd'),
          datetime: fecha.toISOString()
        });
      } else if (/Resumen tarjeta/i.test(asunto)) {
        const vence = texto.match(/vencimiento\s+(\d{1,2})\s+([A-Za-zÁÉÍÓÚáéíóú]{3})[A-Za-zÁÉÍÓÚáéíóú]*\.?\s+(\d{4})/i);
        const pesos = texto.match(/Total en pesos\s*\$\s*([\d.,]+)/i);
        const dolares = texto.match(/Total en d[oó]lares\s*(?:USD|US\$)\s*([\d.,]+)/i);
        if (!vence || !pesos) return;
        const mes = MESES[vence[2].toLowerCase()];
        if (!mes) return;
        resumenes.push({
          month: vence[3] + '-' + mes,
          due_date: vence[3] + '-' + mes + '-' + ('0' + vence[1]).slice(-2),
          total: numero(pesos[1]),
          total_usd: dolares ? numero(dolares[1]) : 0
        });
      }
    });
  });
  const respuesta = UrlFetchApp.fetch(APP_URL + '/rest/v1/rpc/import_uala', {
    method: 'post',
    contentType: 'application/json',
    headers: { apikey: APP_KEY },
    payload: JSON.stringify({ p_token: MI_CODIGO, p_transfers: transferencias, p_statements: resumenes }),
    muteHttpExceptions: true
  });
  Logger.log('Encontré ' + transferencias.length + ' transferencias y ' + resumenes.length + ' resúmenes. Respuesta: ' + respuesta.getResponseCode() + ' ' + respuesta.getContentText());
}

const MESES = { ene: '01', jan: '01', feb: '02', mar: '03', abr: '04', apr: '04', may: '05', jun: '06', jul: '07', ago: '08', aug: '08', sep: '09', set: '09', oct: '10', nov: '11', dic: '12', dec: '12' };

// "$50.000,00" o "$3500" → 50000 / 3500
function numero(texto) {
  const limpio = texto.indexOf(',') >= 0 ? texto.replace(/\./g, '').replace(',', '.') : texto.replace(/\.(?=\d{3}(\D|$))/g, '');
  return Number(limpio);
}

function titulo(texto) {
  return texto.toLowerCase().replace(/(^|\s)\S/g, function (letra) { return letra.toUpperCase(); });
}
